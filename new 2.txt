    <script>
    jQuery(document).ready(function($) {
        var faqIndex = 4;
        
        // افزودن سوال جدید
        $('#boom-add-faq-item').on('click', function() {
            var newItem = `
                <div class="boom-faq-item" data-index="${faqIndex}">
                    <div class="boom-faq-item-header">
                        <span class="boom-faq-drag-handle dashicons dashicons-menu"></span>
                        <strong>سوال ${faqIndex + 1}</strong>
                        <button type="button" class="button boom-faq-remove-item" title="حذف">×</button>
                    </div>
                    <div class="boom-faq-item-content">
                        <p>
                            <label>سوال:</label>
                            <input type="text" name="boom_product_faq[${faqIndex}][question]" value="" class="widefat" placeholder="سوال را وارد کنید...">
                        </p>
                        <p>
                            <label>پاسخ:</label>
                            <textarea name="boom_product_faq[${faqIndex}][answer]" class="widefat" rows="3" placeholder="پاسخ را وارد کنید..."></textarea>
                        </p>
                    </div>
                </div>
            `;
            $('#boom-faq-items').append(newItem);
            faqIndex++;
        });
        
        // حذف سوال
        $(document).on('click', '.boom-faq-remove-item', function() {
            $(this).closest('.boom-faq-item').slideUp(200, function() {
                $(this).remove();
                updateFaqNumbers();
            });
        });
        
        // بروزرسانی شماره‌گذاری
        function updateFaqNumbers() {
            $('#boom-faq-items .boom-faq-item').each(function(index) {
                $(this).find('.boom-faq-item-header strong').text('سوال ' + (index + 1));
            });
        }
        
        // Sortable برای drag & drop
        if (typeof $.fn.sortable !== 'undefined') {
            $('#boom-faq-items').sortable({
                handle: '.boom-faq-drag-handle',
                placeholder: 'boom-faq-placeholder',
                update: function() {
                    updateFaqNumbers();
                }
            });
        }
    });
    </script>
