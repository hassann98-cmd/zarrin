    <script>
        (function() {
            var input = document.getElementById('boom-adv-json');
            var wrap = document.getElementById('boom-adv-groups');

            // کدگذاری/کدگشایی base64 با پشتیبانی UTF-8 (فارسی)
            function b64Encode(str) { return btoa(unescape(encodeURIComponent(str))); }
            function b64Decode(str) { return decodeURIComponent(escape(atob(str))); }

            function parseInit(v) {
                v = v || '';
                if (!v) return [];
                // ابتدا base64
                try { var d = JSON.parse(b64Decode(v)); if (Array.isArray(d)) return d; } catch (e) {}
                // سازگاری با نسخه‌ی قبلی (JSON خام)
                try { var d2 = JSON.parse(v); if (Array.isArray(d2)) return d2; } catch (e) {}
                return [];
            }

            var data = parseInit(input.value);

            function sync() {
                input.value = b64Encode(JSON.stringify(data));
            }
            // فیلد مخفی را از همان ابتدا با base64 هماهنگ کن
            sync();

            function render() {
                wrap.innerHTML = '';
                data.forEach(function(group, gi) {
                    var g = document.createElement('div');
                    g.className = 'boom-adv-group';

                    var head = document.createElement('div');
                    head.className = 'boom-adv-group-head';
                    var gTitle = document.createElement('input');
                    gTitle.type = 'text';
                    gTitle.placeholder = 'عنوان گروه (مثلاً: حافظه داخلی)';
                    gTitle.value = group.title || '';
                    gTitle.addEventListener('input', function() { data[gi].title = gTitle.value; sync(); });
                    var gDel = document.createElement('button');
                    gDel.type = 'button';
                    gDel.className = 'button boom-adv-btn-danger';
                    gDel.textContent = 'حذف گروه';
                    gDel.addEventListener('click', function() { data.splice(gi, 1); render(); sync(); });
                    head.appendChild(gTitle);
                    head.appendChild(gDel);
                    g.appendChild(head);

                    (group.items || []).forEach(function(item, ii) {
                        var row = document.createElement('div');
                        row.className = 'boom-adv-item';

                        // آیکون (SVG / اموجی / لینک تصویر) - اختیاری
                        var iconWrap = document.createElement('div');
                        iconWrap.className = 'boom-adv-icon-wrap';
                        var preview = document.createElement('div');
                        preview.className = 'boom-adv-icon-preview';
                        var updatePreview = function(v) {
                            v = (v || '').trim();
                            if (!v) { preview.innerHTML = ''; return; }
                            if (/^https?:\/\//i.test(v) || /\.(png|jpe?g|webp|svg|gif)$/i.test(v)) {
                                preview.innerHTML = '<img src="' + v.replace(/"/g, '&quot;') + '" alt="" />';
                            } else {
                                preview.innerHTML = v; // SVG یا اموجی
                            }
                        };
                        var icon = document.createElement('textarea');
                        icon.className = 'boom-adv-icon';
                        icon.rows = 1;
                        icon.placeholder = 'آیکون (SVG / اموجی / لینک) - اختیاری';
                        icon.value = item.icon || '';
                        icon.addEventListener('input', function() { data[gi].items[ii].icon = icon.value; updatePreview(icon.value); sync(); });
                        updatePreview(icon.value);
                        iconWrap.appendChild(icon);
                        iconWrap.appendChild(preview);

                        var label = document.createElement('input');
                        label.type = 'text';
                        label.placeholder = 'عنوان ویژگی (مثلاً: اس اس دی)';
                        label.value = item.label || '';
                        label.addEventListener('input', function() { data[gi].items[ii].label = label.value; sync(); });
                        var value = document.createElement('input');
                        value.type = 'text';
                        value.placeholder = 'مقدار (مثلاً: ۵۰۰ گیگ سامسونگ)';
                        value.value = item.value || '';
                        value.addEventListener('input', function() { data[gi].items[ii].value = value.value; sync(); });
                        var del = document.createElement('button');
                        del.type = 'button';
                        del.className = 'boom-adv-btn-danger';
                        del.textContent = '×';
                        del.addEventListener('click', function() { data[gi].items.splice(ii, 1); render(); sync(); });
                        row.appendChild(iconWrap);
                        row.appendChild(label);
                        row.appendChild(value);
                        row.appendChild(del);
                        g.appendChild(row);
                    });

                    var addItem = document.createElement('button');
                    addItem.type = 'button';
                    addItem.className = 'button';
                    addItem.textContent = 'افزودن ویژگی';
                    addItem.addEventListener('click', function() {
                        if (!data[gi].items) data[gi].items = [];
                        data[gi].items.push({ label: '', value: '' });
                        render(); sync();
                    });
                    g.appendChild(addItem);

                    wrap.appendChild(g);
                });
            }

            document.getElementById('boom-adv-add-group').addEventListener('click', function() {
                data.push({ title: '', items: [{ label: '', value: '' }] });
                render(); sync();
            });

            render();
        })();
    </script>
